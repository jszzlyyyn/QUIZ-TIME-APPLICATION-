        static void showQuestion() {
        JFrame f = new JFrame("QUIZ GAME");
        f.setSize(515, 600);
        f.setLayout(null);
        f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        f.setResizable(false);
        f.setLocationRelativeTo(null);
        f.getContentPane().setBackground(BG_DARK);

        JPanel topBar = new JPanel(null);
        topBar.setBounds(0, 0, 500, 55);
        topBar.setBackground(BG_PANEL);
        f.add(topBar);

        JLabel qNumLbl = new JLabel("Q " + (currentQuestion + 1) + " of " + questions.length);
        qNumLbl.setBounds(15, 14, 180, 25);
        qNumLbl.setFont(new Font("Arial", Font.BOLD, 14));
        qNumLbl.setForeground(BLUE);
        topBar.add(qNumLbl);

        JLabel scoreLbl = new JLabel("Score: " + score, SwingConstants.RIGHT);
        scoreLbl.setBounds(300, 14, 185, 25);
        scoreLbl.setFont(new Font("Arial", Font.BOLD, 14));
        scoreLbl.setForeground(WHITE);
        topBar.add(scoreLbl);

        JPanel timerBg = new JPanel(null);
        timerBg.setBounds(0, 55, 500, 8);
        timerBg.setBackground(BG_PANEL);
        f.add(timerBg);

        JPanel timerFill = new JPanel();
        timerFill.setBounds(0, 0, 500, 8);   // starts full
        timerFill.setBackground(GREEN);
        timerBg.add(timerFill);

       
        JLabel timerLbl = new JLabel("15", SwingConstants.CENTER);
        timerLbl.setBounds(215, 70, 70, 40);
        timerLbl.setFont(new Font("Arial", Font.BOLD, 26));
        timerLbl.setForeground(GREEN);
        timerLbl.setBackground(BG_PANEL);
        timerLbl.setOpaque(true);
        f.add(timerLbl);

        JLabel timerTag = new JLabel("SEC", SwingConstants.CENTER);
        timerTag.setBounds(215, 108, 70, 18);
        timerTag.setFont(new Font("Arial", Font.PLAIN, 11));
        timerTag.setForeground(GRAY_TEXT);
        f.add(timerTag);
       

        JLabel questionBox = new JLabel(
            "<html><div style='text-align:center'>" + questions[currentQuestion] + "</div></html>",
            SwingConstants.CENTER);
        questionBox.setBounds(20, 135, 460, 80);
        questionBox.setFont(new Font("Arial", Font.BOLD, 16));
        questionBox.setForeground(WHITE);
        questionBox.setBackground(BG_PANEL);
        questionBox.setOpaque(true);
        f.add(questionBox);
       
        JLabel feedback = new JLabel("", SwingConstants.CENTER);
        feedback.setBounds(20, 222, 460, 28);
        feedback.setFont(new Font("Arial", Font.BOLD, 14));
        f.add(feedback);

      
        String[] labels = {"A", "B", "C", "D"};
        int[] yPos       = {260, 330, 400, 470};
        JButton[] optBtns = new JButton[4];

        for (int i = 0; i < 4; i++) {
            JButton btn = new JButton(choices[currentQuestion][i]);
            btn.setBounds(20, yPos[i], 460, 52);
            btn.setBackground(BG_PANEL);
            btn.setForeground(WHITE);
            btn.setFont(new Font("Arial", Font.PLAIN, 15));
            btn.setFocusPainted(false);
            btn.setBorderPainted(false);
            btn.setHorizontalAlignment(SwingConstants.LEFT);
            btn.setCursor(new Cursor(Cursor.HAND_CURSOR));
            f.add(btn);
            optBtns[i] = btn;

            final String picked = labels[i];
            final int    idx    = i;

                       btn.addMouseListener(new MouseAdapter() {
                public void mouseEntered(MouseEvent e) {
                    if (btn.isEnabled()) btn.setBackground(new Color(60, 60, 90));
                }
                public void mouseExited(MouseEvent e) {
                    if (btn.isEnabled()) btn.setBackground(BG_PANEL);
                }
            });
 
           btn.addActionListener(e -> {
                if (countdown != null) countdown.stop();
                for (JButton b : optBtns) b.setEnabled(false);

                boolean correct = picked.equals(answers[currentQuestion]);
                if (correct) {
                    score++;
                    optBtns[idx].setBackground(GREEN);
                    feedback.setText("Correct!");
                    feedback.setForeground(GREEN);
                } else {
                    optBtns[idx].setBackground(RED_CLR);
                    feedback.setText("Wrong!  Correct answer: " + answers[currentQuestion]);
                    feedback.setForeground(RED_CLR);
                    for (int j = 0; j < 4; j++) {
                        if (labels[j].equals(answers[currentQuestion]))
                            optBtns[j].setBackground(GREEN);
                    }

                }

                Timer next = new Timer(1300, ev -> {
                    f.dispose();
                    currentQuestion++;
                    if (currentQuestion < questions.length) showQuestion();
                    else showResults();
                });
                next.setRepeats(false);
                next.start();
            });
        }

        f.setVisible(true);

        timeLeft = 15;
        countdown = new Timer(1000, null);
        countdown.addActionListener(e -> {
            timeLeft--;
            timerLbl.setText(String.valueOf(timeLeft));

            // Update fill bar width proportionally
            int fillW = (int)((timeLeft / 15.0) * 500);
            timerFill.setBounds(0, 0, fillW, 8);

            // Color shifts as time runs low
            Color barColor = timeLeft > 8 ? GREEN : timeLeft > 4 ? YELLOW : RED_CLR;

            timerFill.setBackground(barColor);
            timerLbl.setForeground(barColor);

            if (timeLeft <= 0) {
                countdown.stop();
                for (JButton b : optBtns) b.setEnabled(false);
                feedback.setText("Time's up!  Answer: " + answers[currentQuestion]);
                feedback.setForeground(YELLOW);
                for (int j = 0; j < 4; j++) {
                    if (labels[j].equals(answers[currentQuestion]))
                        optBtns[j].setBackground(GREEN);
                }
                Timer next = new Timer(1300, ev -> {
                    f.dispose();
                    currentQuestion++;
                    if (currentQuestion < questions.length) showQuestion();
                    else showResults();
                });

                next.setRepeats(false);
                next.start();
            }
        });
        countdown.start();
    }

       static void showResults() {
        JFrame f = new JFrame("QUIZ GAME - Results");
        f.setSize(515, 500);
        f.setLayout(null);
        f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        f.setResizable(false);
        f.setLocationRelativeTo(null);
        f.getContentPane().setBackground(BG_DARK);

        int pct = (score * 100) / questions.length;
        String grade, gradeColor;
        Color  gradeClr;
        if      (pct == 100) { grade = "PERFECT!";         gradeClr = YELLOW; }
        else if (pct >= 75)  { grade = "GREAT JOB!";       gradeClr = GREEN;  }
        else if (pct >= 50)  { grade = "GOOD EFFORT!";     gradeClr = BLUE;   }
        else                 { grade = "KEEP PRACTICING!"; gradeClr = RED_CLR;}

        String msg = pct == 100 ? "You answered everything correctly!"
                   : pct >= 75  ? "Almost perfect — well done!"
                   : pct >= 50  ? "More than half right, keep it up!"
                   :              "Practice more and try again!";

        JLabel gradeLbl = new JLabel(grade, SwingConstants.CENTER);
        gradeLbl.setBounds(0, 70, 500, 55);
        gradeLbl.setFont(new Font("Arial", Font.BOLD, 36));
        gradeLbl.setForeground(gradeClr);
        f.add(gradeLbl);
       
        JPanel scorePanel = new JPanel(null);
        scorePanel.setBounds(100, 145, 300, 110);
        scorePanel.setBackground(BG_PANEL);
        f.add(scorePanel);

        JLabel scoreBig = new JLabel(score + " / " + questions.length, SwingConstants.CENTER);

        scoreBig.setForeground(WHITE);
        scorePanel.add(scoreBig);

        JLabel pctLbl = new JLabel(pct + "% correct", SwingConstants.CENTER);
        pctLbl.setBounds(0, 65, 300, 25);
        pctLbl.setFont(new Font("Arial", Font.PLAIN, 14));
        pctLbl.setForeground(GRAY_TEXT);
        scorePanel.add(pctLbl);

        JLabel msgLbl = new JLabel(msg, SwingConstants.CENTER);
        msgLbl.setBounds(0, 270, 500, 25);
        msgLbl.setFont(new Font("Arial", Font.ITALIC, 14));
        msgLbl.setForeground(GRAY_TEXT);
        f.add(msgLbl);

        again.setBounds(55, 320, 175, 48);

        again.setFocusPainted(false);
        again.setBorderPainted(false);
        again.setCursor(new Cursor(Cursor.HAND_CURSOR));
        f.add(again);

        JButton menu = new JButton("Main Menu");
        menu.setBounds(265, 320, 175, 48);
        menu.setBackground(BG_PANEL);
        menu.setForeground(WHITE);
        menu.setFont(new Font("Arial", Font.BOLD, 15));
        menu.setFocusPainted(false);
        menu.setBorderPainted(false);
        menu.setCursor(new Cursor(Cursor.HAND_CURSOR));
        f.add(menu);

        quit.setForeground(RED_CLR);

        quit.setBorderPainted(false);
        quit.setCursor(new Cursor(Cursor.HAND_CURSOR));
        f.add(quit);
